\begin{abstract}
Threshold secret sharing is a foundational primitive for distributed key management and confidential AI agent pipelines. However, contemporary production implementations (e.g., \texttt{c0dearm/sharks}, \texttt{snipsco/rust-threshold-secret-sharing}) exhibit critical microarchitectural side-channel vulnerabilities, including cache-timing leakage via $GF(2^8)$ lookup tables (\texttt{LOG\_TABLE}) and branch misprediction timing leaks in variable-time modular arithmetic. In this work, we introduce \textbf{Aegis-SSS}, a constant-time threshold secret sharing scheme over the Mersenne prime field $\mathbb{F}_{2^{31}-1}$. By combining branchless Mersenne reduction, fixed Montgomery ladder inversion via Fermat's Little Theorem, and linear-algebraic Cauchy-Lagrange dual basis reconstruction, Aegis-SSS eliminates all memory lookup tables and secret-dependent branches. We formalize the algebraic security properties under the Maximum Distance Separable (MDS) code framework and empirically validate side-channel immunity using Dudect Welch's $t$-test ($|t| < 4.5$).
\end{abstract}

\section{Introduction and Threat Model}
Modern autonomous AI agents require access to high-value Master API Keys (OpenAI, AWS Bedrock, Google Gemini). Under prompt injection attacks or host-provider logging, credentials residing in agent context memory are compromised. While threshold secret sharing $(k, n)$ enables splitting master keys into shares, conventional software libraries introduce catastrophic physical side-channels:
\begin{enumerate}
    \item \textbf{Cache-Timing Attacks (Prime+Probe / Flush+Reload):} In \texttt{c0dearm/sharks}, field multiplication in $\mathbb{F}_{2^8}$ accesses precomputed arrays $\mathtt{LOG\_TABLE}[256]$ and $\mathtt{EXP\_TABLE}[512]$. Cache line evictions directly reveal secret polynomial coefficients.
    \item \textbf{Branch Misprediction Timing Attacks:} In \texttt{snipsco}, inversion relies on the extended Euclidean algorithm with secret-dependent while-loops and conditional branches, creating observable microsecond timing differentials.
\end{enumerate}

\section{Algebraic Formulation}

\subsection{Mersenne Prime Field $\mathbb{F}_{2^{31}-1}$}
Let $p = 2^{31} - 1 = 2147483647$ be the 8th Mersenne prime ($M_{31}$). 

\begin{definition}[Constant-Time Fast Mersenne Reduction]
For any integer product $z = a \cdot b$ with $0 \le a, b < p$, we have $z < 2^{62}$. Expressing $z = 2^{31} q + r$, the modular reduction satisfies:
\begin{equation}
z \equiv q + r \pmod{2^{31} - 1}
\end{equation}
\end{definition}

In Aegis-SSS, this reduction is executed strictly using bitwise shifts and masks:
\begin{algorithmic}[1]
\State $low \gets z \land (2^{31} - 1)$
\State $high \gets z \gg 31$
\State $sum \gets low + high$
\State \textbf{return} $\text{conditional\_select}(sum, sum - p, sum \ge p)$
\end{algorithmic}
This requires zero division instructions, zero hardware divider stalls, and zero secret-dependent control flow.

\subsection{Constant-Time Inversion via Fermat's Little Theorem}
\begin{theorem}
For any non-zero element $x \in \mathbb{F}_p^*$, the multiplicative inverse is:
\begin{equation}
x^{-1} \equiv x^{p-2} = x^{2^{31}-3} \pmod p
\end{equation}
\end{theorem}

\begin{proof}
By Fermat's Little Theorem, $x^{p-1} \equiv 1 \pmod p$. Multiplying both sides by $x^{-1}$ yields $x^{p-2} \equiv x^{-1} \pmod p$.
\end{proof}

The exponent $2^{31} - 3 = (0x7FFFFFD)_{16} = (1111111111111111111111111111101)_2$ contains exactly 30 ones and 1 zero. We implement an addition chain with exactly 30 squarings and 29 constant-time conditional multiplications:
\begin{equation}
\text{Total Operations} = 30 \times \mathtt{square\_ct} + 29 \times \mathtt{mul\_ct} \equiv \text{constant cycles}
\end{equation}

\subsection{Cauchy Matrix MDS Properties}
Let $X = \{x_1, \dots, x_n\}$ and $Y = \{y_1, \dots, y_k\}$ be disjoint subsets of $\mathbb{F}_p$.
\begin{definition}[Cauchy Matrix]
The Cauchy matrix $C \in \mathbb{F}_p^{n \times k}$ has entries:
\begin{equation}
C_{i, j} = \frac{1}{x_i - y_j} \pmod p
\end{equation}
\end{definition}

By the Cauchy determinant formula:
\begin{equation}
\det(C) = \frac{\prod_{1 \le i < j \le k} (x_j - x_i)(y_i - y_j)}{\prod_{1 \le i, j \le k} (x_i - y_j)} \ne 0
\end{equation}
Thus, every $k \times k$ submatrix is strictly non-singular. Any $k$ shares uniquely and non-interactively reconstruct the master secret $S = f(0)$ using dual basis Lagrange weights:
\begin{equation}
S = \sum_{j=1}^k y_j \prod_{m \ne j} \frac{-x_m}{x_j - x_m} \pmod p
\end{equation}

\section{Side-Channel Resistance and Dudect Verification}
We subject Aegis-SSS to the Dudect differential computation analysis framework using Welch's two-sample $t$-test across $N = 500$ interleaved measurements of fixed secrets ($\mathcal{D}_0$) versus random secrets ($\mathcal{D}_1$):
\begin{equation}
t = \frac{\bar{X}_0 - \bar{X}_1}{\sqrt{\frac{s_0^2}{N_0} + \frac{s_1^2}{N_1}}}
\end{equation}
Under the null hypothesis $H_0$ (distributions are indistinguishable), $|t| < 4.5$ proves zero statistically observable timing leakage.
